/*
 * WireMock test double for Stripe.js (https://js.stripe.com/v3/, https://js.stripe.com/<release>/stripe.js).
 *
 * Load this from the mock API with a blocking <script> tag placed before the app bundle:
 *
 *   <script src="https://<mock-host>/stripe-js/v3/"></script>
 *
 * (or https://<mock-host>/stripe-js/<release>/stripe.js, e.g. /stripe-js/endive/stripe.js, which also
 * makes Stripe.version match what a recent @stripe/stripe-js expects; ?version=<release> does the same).
 *
 * @stripe/stripe-js's loadStripe() uses a pre-existing window.Stripe instead of injecting the real
 * script, so the app then talks to this double and to the mock API, never to js.stripe.com or
 * api.stripe.com. An app with a Content-Security-Policy must allow the mock origin in script-src
 * and connect-src.
 *
 * Nothing here looks like Stripe's UI. Elements render plain placeholders in the page (no iframes) and
 * tests drive them through outcomes, found by stable attributes.
 *
 * Payments (Payment Element, Card Element):
 *   - elements.create('payment' | 'card' | 'cardNumber') renders
 *       <div data-mock-element="payment" data-mock-state="ready" data-mock-selected-outcome="succeed">
 *     containing one radio input per outcome:
 *       <input type="radio" data-mock-outcome="succeed | decline | insufficient_funds |
 *              requires_action_succeed | requires_action_fail | incomplete">
 *     The outcome belongs to the elements group. Its default is "succeed"; change the default without
 *     UI with window.__stripeMock = {outcome: 'decline'} or ?outcome=decline on the script URL.
 *   - stripe.confirmPayment / confirmCardPayment send the test card for the selected outcome to the
 *     mock's POST /v1/payment_intents/{id}/confirm with the publishable key and the client secret, so
 *     the PaymentIntent, charge, balance and webhooks behave as they would for that card.
 *   - "requires_action_*" shows <div data-mock-component="authentication"> with buttons
 *     data-mock-outcome="complete" and "fail". It resolves by itself after a short delay (complete,
 *     or fail for requires_action_fail) unless window.__stripeMock.manualAuthentication is true.
 *   - Also: elements.submit, retrievePaymentIntent, handleNextAction, handleCardAction,
 *     createPaymentMethod, createToken, confirmSetup, confirmCardSetup, retrieveSetupIntent
 *     (setup declines are produced by the double itself; the mock's SetupIntents always succeed).
 *
 * Embedded Checkout:
 *   - stripe.initEmbeddedCheckout({clientSecret | fetchClientSecret, onComplete}) resolves
 *     {mount, unmount, destroy}. mount renders <div data-mock-component="embedded-checkout"> with the
 *     session summary and <button data-mock-outcome="pay">. Pay completes the session on the mock
 *     (same result as the hosted page's Pay), then calls onComplete or goes to the session's return_url.
 *   - stripe.redirectToCheckout({sessionId}) goes to the mock's hosted Checkout page.
 *
 * Issuing Elements:
 *   - stripe.createEphemeralKeyNonce({issuingCard}) asks the mock for a nonce.
 *   - elements.create('issuingCardNumberDisplay' | 'issuingCardExpiryDisplay' | 'issuingCardCvcDisplay',

 *       <span data-mock-element="issuingCardNumberDisplay" data-mock-card="ic_..." data-mock-state="ready">

        if (overlay.parentNode) overlay.parentNode.removeChild(overlay);
        resolve(result);
      }
      [['complete', 'Complete authentication'], ['fail', 'Fail authentication']].forEach(function (pair) {
        var button = node('button', {type: 'button', 'data-mock-outcome': pair[0], style: 'margin-right:8px;padding:6px 10px;cursor:pointer;'}, pair[1]);
        button.addEventListener('click', function () { finish(pair[0]); });
        box.appendChild(button);
      });
      overlay.appendChild(box);
      (document.body || document.documentElement).appendChild(overlay);
      if (!settings().manualAuthentication) {
        setTimeout(function () { finish(outcome === 'requires_action_fail' ? 'fail' : 'complete'); }, 700);
      }
    });
  };
  // Takes an intent that may need authentication to its final state. Resolves {intent} or {error}.
  MockStripe.prototype._finish = function (kind, intent, clientSecret, outcome, handleActions) {
    var self = this;
    if (intent.status !== 'requires_action' || handleActions === false) return Promise.resolve({intent: intent});
    var base = '/v1/' + kind + 's/' + encodeURIComponent(intent.id);
    return this._authenticate(outcome).then(function (result) {
      if (result === 'complete') {
        return api(self, 'POST', base + '/confirm', self._key, {client_secret: clientSecret, use_stripe_sdk: 'true', key: self._key})
          .then(function (confirmed) { return {intent: confirmed}; });
      }
      return api(self, 'POST', '/v1/test_helpers/' + kind + 's/' + encodeURIComponent(intent.id) + '/fail_authentication', self._key, {client_secret: clientSecret, key: self._key})
        .then(function (failed) {
          var error = failed.last_payment_error || {type: 'invalid_request_error', code: 'payment_intent_authentication_failure',
            message: 'We are unable to authenticate your payment method. Please choose a different payment method and try again.'};
          error = JSON.parse(JSON.stringify(error));
          error.payment_intent = failed;
          return {error: error};
        });
    }).catch(function (error) { return {error: asError(error)}; });
  };
  MockStripe.prototype._confirmIntent = function (clientSecret, form, outcome, handleActions) {
    var self = this;
    form.client_secret = clientSecret;
    form.use_stripe_sdk = 'true';
    form.key = this._key;
    return api(this, 'POST', '/v1/payment_intents/' + encodeURIComponent(intentId(clientSecret)) + '/confirm', this._key, form).then(
      function (intent) { return self._finish('payment_intent', intent, clientSecret, outcome, handleActions); },
      function (error) { return {error: asError(error)}; }
    );
  };
  function missingSecret(method) {
    return Promise.resolve({error: {type: 'invalid_request_error', code: 'parameter_missing', param: 'clientSecret',
      message: 'stripe.' + method + '(): a clientSecret is required (pass it here or to stripe.elements()).'}});
  }
  MockStripe.prototype.confirmPayment = function (options) {
    options = options || {};
    var group = groupOf(options.elements);
    var clientSecret = options.clientSecret || (group && group._options.clientSecret);
    if (!clientSecret) return missingSecret('confirmPayment');
    var params = options.confirmParams || {};
    var outcome = group ? group._outcome : defaultOutcome();
    if (outcome === 'incomplete' && !params.payment_method) return Promise.resolve({error: INCOMPLETE});
    var data = params.payment_method_data || {};
    var form = params.payment_method ? {payment_method: params.payment_method} : cardFields(outcome, 'payment_method_data', data.billing_details);
    form.expected_payment_method_type = 'card';
    if (params.return_url) form.return_url = params.return_url;
    if (params.receipt_email) form.receipt_email = params.receipt_email;
    if (params.shipping) flatten(params.shipping, 'shipping', form);
    return this._confirmIntent(clientSecret, form, outcome).then(function (result) {
      if (result.error) return result;
      var intent = result.intent;
      if (options.redirect === 'if_required' || !params.return_url) return {paymentIntent: intent};
      // Like Stripe.js: go to return_url and never resolve.
      var url = new URL(params.return_url, window.location.href);
      url.searchParams.set('payment_intent', intent.id);
      url.searchParams.set('payment_intent_client_secret', intent.client_secret || clientSecret);
      url.searchParams.set('redirect_status', intent.status === 'succeeded' || intent.status === 'processing' || intent.status === 'requires_capture' ? 'succeeded' : 'failed');
      window.location.assign(url.toString());
      return new Promise(function () {});
    });
  };
  MockStripe.prototype.confirmCardPayment = function (clientSecret, data, options) {
    data = data || {};
    if (!clientSecret) return missingSecret('confirmCardPayment');
    var method = data.payment_method;
    var form = {};
    var outcome = defaultOutcome();
    if (typeof method === 'string') {
      form.payment_method = method;
    } else if (method && typeof method === 'object') {
      var group = groupOf(method.card);
      if (group) outcome = group._outcome;
      if (outcome === 'incomplete') return Promise.resolve({error: INCOMPLETE});
      if (method.card && method.card.token) form = flatten({type: 'card', card: {token: method.card.token}, billing_details: method.billing_details || {}}, 'payment_method_data', {});
      else form = cardFields(outcome, 'payment_method_data', method.billing_details);
    }
    if (data.return_url) form.return_url = data.return_url;
    if (data.receipt_email) form.receipt_email = data.receipt_email;
    if (data.setup_future_usage) form.setup_future_usage = data.setup_future_usage;
    if (data.shipping) flatten(data.shipping, 'shipping', form);
    return this._confirmIntent(clientSecret, form, outcome, options && options.handleActions).then(function (result) {
      return result.error ? result : {paymentIntent: result.intent};
    });
  };

  MockStripe.prototype.createToken = function (source, data) {
    var group = groupOf(source);

  MockStripe.prototype._confirmSetup = function (clientSecret, outcome, paymentMethod) {
    var self = this;
    if (outcome === 'incomplete' && !paymentMethod) return Promise.resolve({error: INCOMPLETE});
    if (!paymentMethod && (outcome === 'decline' || outcome === 'insufficient_funds' || outcome === 'requires_action_fail')) {
      return this.retrieveSetupIntent(clientSecret).then(function (result) {
        if (result.error) return result;
        var failed = outcome === 'requires_action_fail';
        return {error: {type: failed ? 'invalid_request_error' : 'card_error', code: failed ? 'setup_intent_authentication_failure' : 'card_declined',
          decline_code: failed ? undefined : (outcome === 'decline' ? 'generic_decline' : 'insufficient_funds'),
          message: failed ? 'We are unable to authenticate your payment method. Please choose a different payment method and try again.'
            : (outcome === 'decline' ? 'Your card was declined.' : 'Your card has insufficient funds.'),
          setup_intent: result.setupIntent}};
      });
    }
    var form = {client_secret: clientSecret, use_stripe_sdk: 'true', key: this._key};
    if (paymentMethod) form.payment_method = paymentMethod;
    return api(this, 'POST', '/v1/setup_intents/' + encodeURIComponent(intentId(clientSecret)) + '/confirm', this._key, form).then(
      function (intent) { return {setupIntent: intent}; },
      function (error) { return {error: asError(error)}; }
    );
  };

  MockStripe.prototype.initEmbeddedCheckout = function (options) {
    var stripe = this;
    options = options || {};
    var state = {generation: 0, node: null, destroyed: false};
    function secret() {
      if (!state.secret) {
        state.secret = Promise.resolve(options.clientSecret || (options.fetchClientSecret && options.fetchClientSecret()));
      }
      return state.secret;
    }
    function path(clientSecret, suffix) {
      return '/stripe-js/checkout_sessions/' + encodeURIComponent(clientSecret) + (suffix || '');
    }
    function render(root, session, status, message) {
      while (root.children && root.children.length) root.removeChild(root.children[0]);
      root.setAttribute('data-mock-state', status);
      if (session) root.setAttribute('data-mock-session', session.id);
      var title = node('div', {style: 'font-weight:600;margin-bottom:6px;'}, 'Embedded Checkout');
      title.appendChild(node('span', {style: TAG}, 'Stripe.js test double'));
      root.appendChild(title);
      if (message) root.appendChild(node('div', {'data-mock-status': '', style: 'color:#687385;margin-bottom:6px;'}, message));
      if (!session) return;
      (session.line_items || []).forEach(function (item) {
        root.appendChild(node('div', {'data-mock-line-item': ''}, (item.quantity != null ? item.quantity + ' × ' : '') + item.description + ' — ' + money(item.amount_total, session.currency)));
      });
      root.appendChild(node('div', {'data-mock-amount': '', style: 'margin:6px 0;'}, 'Total: ' + money(session.amount_total, session.currency)));
      if (session.status !== 'open') return;
      var button = node('button', {type: 'button', 'data-mock-outcome': 'pay', style: 'padding:8px 12px;cursor:pointer;'},
        (session.submit_type === 'donate' ? 'Donate ' : session.mode === 'subscription' ? 'Subscribe ' : 'Pay ') + money(session.amount_total, session.currency));
      button.addEventListener('click', function () {
        button.disabled = true;
        secret().then(function (clientSecret) {
          return api(stripe, 'POST', path(clientSecret, '/complete'), stripe._key, {client_secret: clientSecret});
        }).then(function (completed) {
          render(root, Object.assign({}, session, {status: completed.status, payment_status: completed.payment_status}), 'complete', 'Payment complete.');
          if (typeof options.onComplete === 'function') return options.onComplete();
          var returnUrl = completed.return_url || session.return_url;
          if (returnUrl && completed.redirect_on_completion !== 'never') {
            window.location.assign(String(returnUrl).split('{CHECKOUT_SESSION_ID}').join(completed.id));
          }
        }).catch(function (error) {
          render(root, session, 'error', error.message);
          root.setAttribute('data-mock-error', error.message);
        });
      });
      root.appendChild(button);
    }
    var checkout = {
      mount: function (target) {
        if (state.destroyed) throw new Error('This Embedded Checkout instance has been destroyed.');
        var generation = ++state.generation;
        whenPresent(target, function () { return generation === state.generation; }, function (container) {
          var root = node('div', {'data-mock-component': 'embedded-checkout', 'data-mock-state': 'loading', style: BOX});
          state.node = root;
          render(root, null, 'loading', 'Loading…');
          container.appendChild(root);
          secret().then(function (clientSecret) {
            return api(stripe, 'GET', path(clientSecret), stripe._key);
          }).then(function (session) {
            if (generation !== state.generation) return;
            render(root, session, session.status === 'open' ? 'ready' : 'complete', session.status === 'open' ? null : 'This session is ' + session.status + '.');
          }).catch(function (error) {
            if (generation !== state.generation) return;
            render(root, null, 'error', error.message);
            root.setAttribute('data-mock-error', error.message);
            warnOnce('embedded-checkout:' + error.message, 'embedded checkout: ' + error.message);
          });
        });
      },
      unmount: function () {
        state.generation++;
        if (state.node && state.node.parentNode) state.node.parentNode.removeChild(state.node);
        state.node = null;
      },
      destroy: function () {
        checkout.unmount();
        state.destroyed = true;
      },
    };
    return Promise.resolve(checkout);
  };
